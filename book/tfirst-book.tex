% A boxed "record" environment for load-bearing verbatim findings.
% NOTE: \colorbox must receive its argument as one complete group, so the
% body is captured in an lrbox first.  Splitting "\colorbox{c}{" across the
% begin-code and end-code of \newenvironment leaves the begin-code
% brace-unbalanced, which TeX mis-parses at definition time and only fails
% on first use -- which is exactly how this survived the empty-chapter build.
\newsavebox{\recordbox}
\newenvironment{record}
  {\par\medskip\noindent\begin{lrbox}{\recordbox}%
   \begin{minipage}{\dimexpr\linewidth-2\fboxsep}%
   \vspace{2pt}\small}
  {\vspace{2pt}\end{minipage}%
   \end{lrbox}%
   \colorbox{panelgrey}{\usebox{\recordbox}}\par\medskip}

\small
\textbf{Status of this book.} The programme it documents \emph{has not proved}
the Clay Millennium Prize problem. It claimed a proof once, in session~S29,
and withdrew that claim in session~S31 on eight independent grounds. This
book records the routes taken, the reasoning behind each, the numerical
programme built to test them, and --- with no less weight --- every wall the
programme hit and every claim it had to retract. Every quantitative statement
here traces to \texttt{results/results.db}, to a session log, or to
\texttt{proofs/claim\_a\_3d\_proof\_attempt.tex}. Where the sources disagree,
the proof document governs, and the disagreement is footnoted.
\end{minipage}
